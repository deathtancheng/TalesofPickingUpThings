<!DOCTYPE html>
<html>
<head>

    <title>2048 游戏</title>
    <style>
        body { margin: 0; background: #fafafa; font-family: sans-serif; }
        canvas { background: #bbada0; display: block; }
        #score { position: absolute; top: 10px; left: 10px; font-size: 24px; }
        #gameover { display: none; position: absolute; width: 100%; height: 100%; background: rgba(0,0,0,0.7); color: white; text-align: center; padding-top: 20%; font-size: 32px; }
        #gameover button { padding: 10px 20px; font-size: 18px; }
    </style>
</head>
<body>
    <div id="score">分数: 0</div>
    <canvas id="game" width="400" height="400"></canvas>
    <div id="gameover">游戏结束<br><button onclick="location.reload()">再试一次</button></div>
    <script>
        const canvas = document.getElementById('game');
        const ctx = canvas.getContext('2d');
        const scoreEl = document.getElementById('score');
        const gameoverEl = document.getElementById('gameover');
        
        const gridSize = 4;
        const tileSize = canvas.width / gridSize;
        let board = Array.from({length: gridSize}, () => Array(gridSize).fill(0));
        let score = 0;
        
        function drawTile(x, y, value) {
            const color = {
                2: '#eee4da',
                4: '#ede0c8',
                8: '#f2b179',
                16: '#f59563',
                32: '#f67c5f',
                64: '#f65e3b',
                128: '#f99667',
                256: '#f8f8ff',
                512: '#f6f6f6',
                1024: '#f5f5f5',
                2048: '#f5f5f5'
            }[value] || '#cdc1b4';
            
            ctx.fillStyle = color;
            ctx.fillRect(x * tileSize, y * tileSize, tileSize, tileSize);
            
            ctx.fillStyle = '#000';
            ctx.font = `${tileSize * 0.5}px sans-serif`;
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText(value, x * tileSize + tileSize/2, y * tileSize + tileSize/2);
        }
        
        function drawBoard() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            for (let y = 0; y < gridSize; y++) {
                for (let x = 0; x < gridSize; x++) {
                    drawTile(x, y, board[y][x]);
                }
            }
        }
        
        function addTile() {
            const empty = [];
            for (let y = 0; y < gridSize; y++) {
                for (let x = 0; x < gridSize; x++) {
                    if (board[y][x] === 0) empty.push([x, y]);
                }
            }
            if (empty.length === 0) return false;
            const [x, y] = empty[Math.floor(Math.random() * empty.length)];
            board[y][x] = Math.random() < 0.9 ? 2 : 4;
            return true;
        }
        
        function mergeTiles(line) {
            let merged = false;
            for (let i = 0; i < line.length - 1; i++) {
                if (line[i] !== 0 && line[i] === line[i+1]) {
                    line[i] *= 2;
                    line[i+1] = 0;
                    score += line[i];
                    merged = true;
                }
            }
            return merged;
        }
        
        function moveTiles(direction) {
            let moved = false;
            const newBoard = JSON.parse(JSON.stringify(board));
            
            if (direction === 'left') {
                for (let y = 0; y < gridSize; y++) {
                    let line = board[y].filter(x => x !== 0);
                    if (mergeTiles(line)) moved = true;
                    for (let i = 0; i < gridSize; i++) {
                        newBoard[y][i] = i < line.length ? line[i] : 0;
                    }
                }
            } else if (direction === 'right') {
                for (let y = 0; y < gridSize; y++) {
                    let line = board[y].filter(x => x !== 0).reverse();
                    if (mergeTiles(line)) moved = true;
                    for (let i = 0; i < gridSize; i++) {
                        newBoard[y][gridSize - 1 - i] = i < line.length ? line[i] : 0;
                    }
                }
            } else if (direction === 'up') {
                for (let x = 0; x < gridSize; x++) {
                    let line = [];
                    for (let y = 0; y < gridSize; y++) line.push(board[y][x]);
                    line = line.filter(x => x !== 0);
                    if (mergeTiles(line)) moved = true;
                    for (let i = 0; i < gridSize; i++) {
                        newBoard[i][x] = i < line.length ? line[i] : 0;
                    }
                }
            } else if (direction === 'down') {
                for (let x = 0; x < gridSize; x++) {
                    let line = [];
                    for (let y = 0; y < gridSize; y++) line.push(board[y][x]);
                    line = line.filter(x => x !== 0).reverse();
                    if (mergeTiles(line)) moved = true;
                    for (let i = 0; i < gridSize; i++) {
                        new  Board[i][x] = i < line.length ? line[i] : 0;
                    }
                }
            }
            
            board = newBoard;
            drawBoard();
            if (moved || addTile()) {
                scoreEl.textContent = `分数: ${score}`;
                if (!addTile()) checkGameOver();
            }
        }
        
        function checkGameOver() {
            for (let y = 0; y < gridSize; y++) {
                for (let x = 0; x < gridSize; x++) {
                    if (board[y][x] === 0) return;
                    if (x < gridSize - 1 && board[y][x] === board[y][x+1]) return;
                    if (y < gridSize - 1 && board[y][x] === board[y+1][x]) return;
                }
            }
            gameoverEl.style.display = 'block';
        }
        
        document.addEventListener('keydown', (e) => {
            if (e.key === 'ArrowLeft') moveTiles('left');
            else if (e.key === 'ArrowUp') moveTiles('up');
            else if (e.key === 'ArrowRight') moveTiles('right');
            else if (e.key === 'ArrowDown') moveTiles('down');
        });
        
        addTile();
        drawBoard();
    </script>
</body>
</html>
